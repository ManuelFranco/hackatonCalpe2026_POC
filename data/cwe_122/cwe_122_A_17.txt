#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main(int argc, char **argv) {
    if (argc != 2) return 1;
    const size_t cap = 8;
    if (strlen(argv[1]) >= cap) return 1;
    char *data = malloc(cap);
    if (!data) return 1;
    strcpy(data, argv[1]);
    puts(data);
    free(data);
    return 0;
}
