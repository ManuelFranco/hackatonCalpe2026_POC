#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main(int argc, char **argv) {
    if (argc != 2) return 1;
    const size_t cap = 14;
    if (strlen(argv[1]) >= cap) return 1;
    char *item = malloc(cap);
    if (!item) return 1;
    strcpy(item, argv[1]);
    puts(item);
    free(item);
    return 0;
}
