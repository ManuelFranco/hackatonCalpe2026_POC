#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main(int argc, char **argv) {
    if (argc != 2) return 1;
    const size_t cap = 10;
    if (strlen(argv[1]) >= cap) return 1;
    char *label = malloc(cap);
    if (!label) return 1;
    strcpy(label, argv[1]);
    puts(label);
    free(label);
    return 0;
}
