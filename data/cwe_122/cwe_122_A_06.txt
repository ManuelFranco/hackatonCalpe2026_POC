#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main(int argc, char **argv) {
    if (argc != 2) return 1;
    const size_t cap = 13;
    if (strlen(argv[1]) >= cap) return 1;
    char *word = malloc(cap);
    if (!word) return 1;
    strcpy(word, argv[1]);
    puts(word);
    free(word);
    return 0;
}
