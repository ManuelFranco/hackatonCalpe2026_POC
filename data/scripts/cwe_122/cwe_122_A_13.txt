#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main(int argc, char **argv) {
    if (argc != 2) return 1;
    const size_t cap = 12;
    if (strlen(argv[1]) >= cap) return 1;
    char *field = malloc(cap);
    if (!field) return 1;
    strcpy(field, argv[1]);
    puts(field);
    free(field);
    return 0;
}
