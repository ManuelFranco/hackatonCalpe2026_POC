#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main(int argc, char **argv) {
    if (argc != 2) return 1;
    const size_t cap = 9;
    if (strlen(argv[1]) >= cap) return 1;
    char *tag = malloc(cap);
    if (!tag) return 1;
    strcpy(tag, argv[1]);
    puts(tag);
    free(tag);
    return 0;
}
