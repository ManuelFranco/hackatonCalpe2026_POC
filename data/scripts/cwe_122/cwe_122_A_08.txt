#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main(int argc, char **argv) {
    if (argc != 2) return 1;
    const size_t cap = 15;
    if (strlen(argv[1]) >= cap) return 1;
    char *entry = malloc(cap);
    if (!entry) return 1;
    strcpy(entry, argv[1]);
    puts(entry);
    free(entry);
    return 0;
}
