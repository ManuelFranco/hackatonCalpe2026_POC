#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main(int argc, char **argv) {
    if (argc != 2) return 1;
    const size_t cap = 10;
    if (strlen(argv[1]) >= cap) return 1;
    char *code = malloc(cap);
    if (!code) return 1;
    strcpy(code, argv[1]);
    puts(code);
    free(code);
    return 0;
}
